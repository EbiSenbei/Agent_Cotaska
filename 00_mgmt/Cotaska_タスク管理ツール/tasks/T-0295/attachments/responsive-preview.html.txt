<script>
const q=new URLSearchParams(location.search), page=q.get('page')||'index.html';
const width=[320,390].includes(Number(q.get('width')))?Number(q.get('width')):390;
const f=document.querySelector('iframe');f.style.width=width+'px';f.src=['index.html','download.html','usage.html','examples.html'].includes(page)?page:'index.html';
document.querySelector('#viewport-status').textContent=`${f.src.split('/').pop()} / CSS viewport ${width}px（iframe表示。スマホ実機ではありません）`;
function measure(){
  const d=f.contentDocument, viewport=d.documentElement.clientWidth;
  const overflowing=Array.from(d.querySelectorAll('h1,h2,h3,.nowrap,.heading-line,.actions,.button,video,.release-card')).filter(e=>{
    const r=e.getBoundingClientRect();return r.width>0&&(r.right>viewport+1||r.left< -1);
  }).map(e=>({tag:e.tagName,text:e.textContent.slice(0,60)}));
  const contentOverflows=Array.from(d.querySelectorAll('body *')).filter(e=>e.getBoundingClientRect().width>0&&e.scrollWidth>e.clientWidth+1).map(e=>({tag:e.tagName,className:e.className,width:e.clientWidth,content:e.scrollWidth,text:e.textContent.slice(0,50)}));
  const imageRatios=Array.from(d.querySelectorAll('.intro-overview img,.screen-card img')).map(e=>{const r=e.getBoundingClientRect(),ratio=Number(e.getAttribute('width'))/Number(e.getAttribute('height'));return{src:e.getAttribute('src'),width:r.width,height:r.height,aspectRatioPreserved:Math.abs(r.width/r.height/ratio-1)<.02};});
  const split=d.querySelector('.split-heading'),heading=split?.children[0]?.getBoundingClientRect(),copy=split?.children[1]?.getBoundingClientRect();
  const aiLayout=split?{columns:f.contentWindow.getComputedStyle(split).gridTemplateColumns,headingAboveCopy:heading.bottom<=copy.top+1}:null;
  document.querySelector('#layout-result').textContent=JSON.stringify({page:d.location.pathname.split('/').pop(),cssViewport:f.contentWindow.innerWidth,viewport,content:d.documentElement.scrollWidth,overflowing,contentOverflows,imageRatios,aiLayout},null,2);
}
f.addEventListener('load',measure);
document.querySelector('#measure').addEventListener('click',measure);
</script></html>
